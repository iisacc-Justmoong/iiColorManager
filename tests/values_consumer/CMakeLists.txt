cmake_minimum_required(VERSION 3.24)
project(ColorValuesConsumer LANGUAGES CXX)
find_package(iiColorManager 0.1.1 CONFIG REQUIRED COMPONENTS Values)
if(TARGET iiColorManager::iiColorManager)
    message(FATAL_ERROR "A Values-only consumer imported the picker library")
endif()
add_executable(values_consumer main.cpp)
target_link_libraries(values_consumer PRIVATE iiColorManager::Values)
enable_testing()
add_test(NAME iiColorManager.ValuesConsumer COMMAND values_consumer)
